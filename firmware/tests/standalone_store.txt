# RemoteFM1 standalone, 0.3.1 (STANDALONE-PLAN.md question 3, firmware/sa/sa_store.c): the data area. 0.3.0's user
# banks in XDATA0 taken over once, in their order; the 16 loops saved with SAVE held (in steps: one flash operation a
# main-loop pass at most), read back at power-on; a save where nothing changed writes nothing, one where one loop
# changed writes only it and the directory; a power cut in the middle of a save (in its data, at the directory's erase,
# at the directory's write) leaves the old loops. tests/run_tests.sh runs it (both builds).
wait 400
# 0.3.0's banks (the arrivals 5, 6 and 4: slot 2 the oldest), as 0.3.0 left them, and a power cycle: taken over
sa oldbank 0 5
sa oldbank 1 6
sa oldbank 2 4
reboot
wait 400
expect sa_ub_migrated 3
expect sa_ubanks 3
sa info 3 0 OLD3 V01
sa info 3 1 OLD1 V01
sa info 3 2 OLD2 V01
expect sa_ub_refused 0
# once only: again, nothing written
reboot
wait 400
expect sa_ub_migrated 0
expect sa_sf_commits 0
expect sa_ubanks 3
# a bank sent now is the fourth
sa dx7bank 4 usb
expect sa_ubanks 4
sa info 4 3 USR4 V01
# three loops (loop 3 of 1,500 events: three sectors), SAVE held: saved in steps
sa loopfill 0 2
sa loopfill 2 1500
sa loopfill 7 300
press SAVE
wait 600
release SAVE
expect sa_save_holds 1
wait 3000
expect sa_sf_busy 0
expect sa_sf_commits 2
expect sa_sf_pass_max 1
expect sa_sf_pass_bytes 1024
expect sa_ub_refused 0
expect store_outside 0
reboot
wait 400
expect sa_sf_loads 3
expect sa_sf_lbad 0
sa loopcheck 0 2
sa loopcheck 2 1500
sa loopcheck 7 300
expect sa_ubanks 4
# nothing changed: nothing written
press SAVE
wait 600
release SAVE
wait 1000
expect sa_sf_same 1
expect sa_sf_erases 0
# one loop changed: it alone (one sector) and the directory: two erases
sa loopfill 0 5
press SAVE
wait 600
release SAVE
wait 1000
expect sa_sf_erases 2
expect sa_sf_commits 1
# the power gone at the third flash operation of a save (loop 3, 1,000 events: its first sector's second write)
sa loopfill 2 1000
sa cut 3
press SAVE
wait 600
release SAVE
wait 2000
expect sa_sf_aborted 1
reboot
wait 400
sa loopcheck 2 1500
sa loopcheck 0 5
sa loopcheck 7 300
# loop 8 (300 events: one sector: an erase and three writes, then the directory's erase and write): the power gone at
# the directory's erase (operation 5), then at its write (operation 6): the old directory is still in charge
sa loopfill 8 300
sa cut 5
press SAVE
wait 600
release SAVE
wait 2000
reboot
wait 400
expect sa_sf_loads 3
sa loopfill 8 300
sa cut 6
press SAVE
wait 600
release SAVE
wait 2000
reboot
wait 400
expect sa_sf_loads 3
sa loopcheck 7 300
# and without a cut: there after a power cycle
sa loopfill 8 300
press SAVE
wait 600
release SAVE
wait 2000
expect sa_sf_commits 1
reboot
wait 400
expect sa_sf_loads 4
sa loopcheck 8 300
sa loopcheck 2 1500
expect sa_ubanks 4
sa info 4 0 OLD3 V01
expect sa_ub_refused 0
expect store_outside 0
